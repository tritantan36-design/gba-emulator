cmake_minimum_required(VERSION 3.22.1)
project(mgba_smoke LANGUAGES C CXX)
option(GBA_HOST_ASAN "TEST ONLY MSVC host AddressSanitizer" OFF)
option(GBA_ROM_FUZZ "TEST ONLY Clang libFuzzer + ASan + UBSan ROM parser" OFF)
if(GBA_ROM_FUZZ)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        message(FATAL_ERROR "ROM fuzz requires Clang")
    endif()
    add_compile_options(-fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer)
endif()
if(GBA_HOST_ASAN AND MSVC)
    set(CMAKE_C_FLAGS_DEBUG "/Od /Zi")
    set(CMAKE_CXX_FLAGS_DEBUG "/Od /Zi")
    add_compile_options(/fsanitize=address)
    add_link_options(/INCREMENTAL:NO)
endif()
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(LIBMGBA_ONLY ON)
set(M_CORE_GBA ON)
set(M_CORE_GB OFF)
set(MINIMAL_CORE 2)
set(ENABLE_EXTRA OFF)
add_compile_definitions(MINIMAL_CORE=2 DISABLE_THREADING)
add_subdirectory("${ROOT}/third_party/mgba" mgba EXCLUDE_FROM_ALL)
if(GBA_ROM_FUZZ)
    # Upstream sets target COMPILE_OPTIONS; apply instrumentation after target creation.
    target_compile_options(mgba PRIVATE -fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer)
    add_executable(rom_fuzz rom_fuzz.cpp)
    target_link_libraries(rom_fuzz PRIVATE mgba)
    target_link_options(rom_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
    return()
endif()
add_executable(mgba_smoke mgba_smoke.cpp)
target_link_libraries(mgba_smoke PRIVATE mgba)
enable_testing()
add_executable(suite_selected_test suite_selected_test.cpp)
target_link_libraries(suite_selected_test PRIVATE mgba)
add_test(NAME mgba_suite_shifter_selected COMMAND suite_selected_test "${ROOT}/core-mgba/src/androidTest/assets/mgba-suite-shifter.gba")
add_executable(peripheral_test peripheral_test.cpp)
target_link_libraries(peripheral_test PRIVATE mgba)
add_test(NAME mgba_peripherals COMMAND peripheral_test "${ROOT}/core-mgba/src/androidTest/assets")
add_test(NAME mgba_homebrew COMMAND mgba_smoke "${ROOT}/core-mgba/src/androidTest/assets/bringup.gba")
add_executable(persistence_test persistence_test.cpp)
target_link_libraries(persistence_test PRIVATE mgba)
add_test(NAME mgba_persistence COMMAND persistence_test "${ROOT}/core-mgba/src/androidTest/assets/bringup.gba" "${ROOT}/test-rom/test-save-v1.sav")
add_executable(player_test player_test.cpp)
target_link_libraries(player_test PRIVATE mgba)
add_test(NAME mgba_player COMMAND player_test "${ROOT}/core-mgba/src/androidTest/assets/bringup.gba")
add_executable(visual_test visual_test.cpp)
target_link_libraries(visual_test PRIVATE mgba)
foreach(pattern color-pattern lcd-pattern)
    add_test(NAME mgba_${pattern} COMMAND visual_test "${ROOT}/core-mgba/src/androidTest/assets/${pattern}.gba" "${ROOT}/core-mgba/src/androidTest/assets/${pattern}.rgba")
endforeach()
