cmake_minimum_required(VERSION 3.22.1)
project(gba_bridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
option(GBA_SANITIZERS "Enable ASan and UBSan for debug/test only" OFF)
option(GBA_UBSAN "Enable UBSan only for debug/test" OFF)
add_compile_options(-fstack-protector-strong)
add_compile_definitions($<$<NOT:$<CONFIG:Debug>>:_FORTIFY_SOURCE=2>)
if(GBA_SANITIZERS)
    if(NOT CMAKE_BUILD_TYPE STREQUAL "Debug")
        message(FATAL_ERROR "Sanitizers are for debug/test builds")
    endif()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
if(GBA_UBSAN)
    if(NOT CMAKE_BUILD_TYPE STREQUAL "Debug")
        message(FATAL_ERROR "UBSan is for debug/test builds")
    endif()
    add_compile_options(-fsanitize=undefined -fsanitize-trap=undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=undefined -fsanitize-trap=undefined)
endif()
add_link_options(-Wl,-z,relro,-z,now,-z,noexecstack -Wl,-z,max-page-size=16384)
set(LIBMGBA_ONLY ON)
set(M_CORE_GBA ON)
set(M_CORE_GB OFF)
set(MINIMAL_CORE 2)
set(ENABLE_EXTRA OFF)
add_compile_definitions(MINIMAL_CORE=2 DISABLE_THREADING)
add_subdirectory("${ROOT}/third_party/mgba" "${CMAKE_CURRENT_BINARY_DIR}/mgba" EXCLUDE_FROM_ALL)
add_subdirectory("${ROOT}/third_party/oboe" "${CMAKE_CURRENT_BINARY_DIR}/oboe" EXCLUDE_FROM_ALL)
get_target_property(OBOE_HEADERS oboe INTERFACE_INCLUDE_DIRECTORIES)
set_property(TARGET oboe PROPERTY INTERFACE_SYSTEM_INCLUDE_DIRECTORIES "${OBOE_HEADERS}")
add_subdirectory("${ROOT}/audio/src/main/cpp" "${CMAKE_CURRENT_BINARY_DIR}/audio")
add_library(gba_bridge SHARED mgba_bridge.cpp jni_bridge.cpp)
target_compile_definitions(gba_bridge PRIVATE $<$<CONFIG:Debug>:GBA_TEST_HOOKS>)
target_link_libraries(gba_bridge PRIVATE mgba gba_audio log android)
target_compile_options(gba_bridge PRIVATE -Wall -Wextra -Werror)
